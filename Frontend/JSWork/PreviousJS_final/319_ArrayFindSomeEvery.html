Array Methods: find, findIndex, some, every

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

  • find(cb) — returns the first element that satisfies the callback, or undefined
  • findIndex(cb) — returns the index of the first match, or -1
  • some(cb) — returns true if at least one element passes
  • every(cb) — returns true only if all elements pass
const nums = [1, 3, 5, 8, 9];
nums.find(n => n % 2 === 0);      // 8
nums.findIndex(n => n % 2 === 0); // 3
nums.some(n => n > 7);            // true
nums.every(n => n > 0);           // true

Array.prototype.myFind = function(cb) {
  for (let i = 0; i < this.length; i++) {
    if (cb(this[i], i, this)) return this[i];
  }
  return undefined;
};

Array.prototype.mySome = function(cb) {
  for (let i = 0; i < this.length; i++) {
    if (cb(this[i], i, this)) return true;
  }
  return false;
};

const nums = [3, 7, 2, 9, 4];
console.log(nums.myFind(n => n % 2 === 0)); // 2
console.log(nums.mySome(n => n > 8));       // true
console.log(nums.mySome(n => n > 100));     // false

function validate(data, requiredRules, optionalRules) {
  const errors = [];
  const allRequired = requiredRules.every(rule => {
    const result = rule(data);
    if (result !== true) { errors.push(result); return false; }
    return true;
  });
  const anyOptional = optionalRules.some(rule => rule(data) === true);
  return { valid: allRequired && anyOptional, errors };
}

const required = [
  d => d.name?.length > 0 || 'Name required',
  d => d.age >= 18        || 'Must be adult',
];
const optional = [
  d => d.email?.includes('@') || 'Valid email preferred',
  d => d.phone?.length > 0   || 'Phone preferred',
];
console.log(validate(data, required, optional)); // { valid: true, errors: [] }
