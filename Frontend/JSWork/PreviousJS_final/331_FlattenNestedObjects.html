Flatten Nested Objects

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

Flattening converts deeply nested objects into a single-level object with dot-notation keys.

{ a: { b: { c: 1 } }, d: 2 }
→ { 'a.b.c': 1, 'd': 2 }

Common in form handling, API response normalization, and config management.

🎯 Interview Challenges

EASY Flatten one level deep

Flatten an object one level — merge nested object keys into parent with underscore separator.

// Starter Code
function flattenOneLevel(obj, separator = '_') {
  const result = {};
  for (const [key, value] of Object.entries(obj)) {
    if (value && typeof value === 'object' && !Array.isArray(value)) {
      for (const [k2, v2] of Object.entries(value)) {
        result[key + separator + k2] = v2;
      }
    } else {
      result[key] = value;
    }
  }
  return result;
}

console.log(flattenOneLevel({ a: { x: 1, y: 2 }, b: 3 }));
// { a_x: 1, a_y: 2, b: 3 }

function flatten(obj, prefix = '', result = {}) {
  for (const [key, value] of Object.entries(obj)) {
    const fullKey = prefix ? `${prefix}.${key}` : key;
    if (value && typeof value === 'object' && !Array.isArray(value))
      flatten(value, fullKey, result);
    else
      result[fullKey] = value;
  }
  return result;
}

console.log(flatten({ a: { b: { c: 1 }, d: 2 }, e: 3, f: { g: { h: { i: 4 } } } }));
// { 'a.b.c': 1, 'a.d': 2, e: 3, 'f.g.h.i': 4 }

function flatten(obj, prefix='', res={}) {
  for (const [k, v] of Object.entries(obj)) {
    const key = prefix ? `${prefix}.${k}` : k;
    if (v && typeof v === 'object' && !Array.isArray(v)) flatten(v, key, res);
    else res[key] = v;
  }
  return res;
}

function unflatten(flat) {
  const result = {};
  for (const [dotKey, value] of Object.entries(flat)) {
    const parts = dotKey.split('.');
    parts.reduce((obj, part, i) => {
      if (i === parts.length - 1) { obj[part] = value; }
      else { obj[part] = obj[part] || {}; }
      return obj[part];
    }, result);
  }
  return result;
}
