Shortest Distance — Bellman-Ford Algorithm

📋 Problem

Find shortest distance from source to all nodes in a weighted directed graph. Unlike Dijkstra, handles negative weights. Detect negative cycles.

// Graph: 0→1 (4), 0→2 (1), 2→1 (2), 1→3 (1)
bellmanFord(4, [[0,1,4],[0,2,1],[2,1,2],[1,3,1]], 0)
// [0, 3, 1, 4]

💡 Approach

Relax all edges V-1 times. On each pass, update dist[v] = min(dist[v], dist[u]+weight). A V-th pass that still relaxes = negative cycle detected.

✅ Solution

function bellmanFord(numNodes, edges, source) {
  const dist = new Array(numNodes).fill(Infinity);
  dist[source] = 0;

  // Relax all edges V-1 times
  for (let i = 0; i < numNodes - 1; i++) {
    let updated = false;
    for (const [u, v, w] of edges) {
      if (dist[u] !== Infinity && dist[u] + w < dist[v]) {
        dist[v] = dist[u] + w;
        updated = true;
      }
    }
    if (!updated) break; // early termination
  }

  // V-th pass: detect negative cycle
  for (const [u, v, w] of edges) {
    if (dist[u] !== Infinity && dist[u] + w < dist[v]) {
      return 'Negative cycle detected';
    }
  }

  return dist.map(d => d === Infinity ? -1 : d);
}

// Test 1: normal graph
console.log(bellmanFord(4, [[0,1,4],[0,2,1],[2,1,2],[1,3,1]], 0));
// [0, 3, 1, 4]

// Test 2: graph with negative edge (no negative cycle)
console.log(bellmanFord(5, [[0,1,6],[0,2,7],[1,2,8],[1,3,-4],[1,4,5],[2,4,-3],[3,0,2],[4,3,7]], 0));

// Test 3: negative cycle
console.log(bellmanFord(3, [[0,1,1],[1,2,-2],[2,0,-1]], 0));
// 'Negative cycle detected'
