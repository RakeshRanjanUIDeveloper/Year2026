Object.freeze() and Object.seal()

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

Both methods restrict object mutation but differ in what they allow:

  • Object.freeze(obj) — No add, no delete, no modify. Fully immutable (shallow).
  • Object.seal(obj) — No add, no delete. Existing properties CAN be modified.
  • Neither is deep — nested objects are still mutable.
const obj = Object.freeze({ x: 1, nested: { y: 2 } });
obj.x = 99;         // silently fails (throws in strict mode)
obj.nested.y = 99;  // WORKS — freeze is shallow!

▶ Show Solution 
function deepFreeze(obj) {
  Object.getOwnPropertyNames(obj).forEach(name => {
    const val = obj[name];
    if (val && typeof val === 'object') deepFreeze(val);
  });
  return Object.freeze(obj);
}

const config = deepFreeze({
  server: { host: 'localhost', port: 3000 },
  db: { name: 'mydb' }
});

config.server.host = 'evil.com';
console.log(config.server.host); // 'localhost' — deep freeze works

▶ Show Solution 
function createStore(initialState) {
  let state = Object.freeze({ ...initialState });

  return {
    getState() { return state; },
    update(partial) {
      state = Object.freeze({ ...state, ...partial });
      return state;
    }
  };
}

const store = createStore({ user: 'Alice', count: 0 });
const s1 = store.getState();
console.log(s1);                // { user: 'Alice', count: 0 }
store.update({ count: 1 });
const s2 = store.getState();
console.log(s2);                // { user: 'Alice', count: 1 }
console.log(s1 === s2);         // false — new object each update
console.log(s1.count);          // 0 — original unchanged
