Optional Chaining (?.) and Nullish Coalescing (??)

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

  • obj?.prop — returns undefined instead of throwing if obj is null/undefined
  • arr?.[i] — safe index access
  • fn?.() — safe function call
  • a ?? b — returns b only if a is null or undefined (unlike || which also triggers on 0, '', false)
const user = null;
user?.address?.city;     // undefined, no error
user?.getName?.();       // undefined, no error
const port = config.port ?? 3000; // 3000 only if port is null/undefined
const port2 = config.port || 3000; // 3000 even if port is 0 — BUG!

🎯 Interview Challenges

EASY ?? vs || — spot the difference

Why does using || instead of ?? cause bugs with falsy values? Show with examples.

// Starter Code
const config = { port: 0, debug: false, name: '' };

// Using || (wrong for falsy values)
const port1   = config.port  || 8080;
const debug1  = config.debug || true;
const name1   = config.name  || 'default';

// Using ?? (correct — only null/undefined triggers fallback)
const port2   = config.port  ?? 8080;
const debug2  = config.debug ?? true;
const name2   = config.name  ?? 'default';

console.log(port1, debug1, name1);  // ?
console.log(port2, debug2, name2);  // ?
▶ Show Solution 
// port1=8080, debug1=true, name1='default'  ← all WRONG! falsy values lost
// port2=0,    debug2=false, name2=''          ← all CORRECT! 0/''/false preserved

// Rule: use ?? when 0, false, or '' are valid values
//       use || only when you want ALL falsy values to fall back
MEDIUM Safe deep access utility

Write a safeGet(obj, path, defaultValue) that safely accesses a nested path like 'user.address.city' using optional chaining logic (implement it manually, without ?.).

// Starter Code
function safeGet(obj, path, defaultValue = undefined) {
  // split path by '.', reduce with safety checks
  // return defaultValue if anything is null/undefined
}

const data = { user: { address: { city: 'Mumbai' } } };
console.log(safeGet(data, 'user.address.city'));     // 'Mumbai'
console.log(safeGet(data, 'user.phone.number', 'N/A')); // 'N/A'
console.log(safeGet(null, 'user.name', 'anon'));     // 'anon'
▶ Show Solution 
function safeGet(obj, path, defaultValue = undefined) {
  const result = path.split('.').reduce((acc, key) => {
    return acc != null ? acc[key] : undefined;
  }, obj);
  return result ?? defaultValue;
}

const data = { user: { address: { city: 'Mumbai' } } };
console.log(safeGet(data, 'user.address.city'));         // 'Mumbai'
console.log(safeGet(data, 'user.phone.number', 'N/A')); // 'N/A'
console.log(safeGet(null, 'user.name', 'anon'));         // 'anon'
HARD Config merger with nullish fallback chain

Build a getConfig() that merges user config → env config → defaults using ??. Each layer only overrides if the value is actually missing (null/undefined), not if it's 0 or false.

// Starter Code
function getConfig(userConfig, envConfig, defaults) {
  // for each key in defaults, pick: user ?? env ?? default
  // must preserve 0, false, '' from user/env
}

const defaults = { port: 3000, debug: false, workers: 4, name: 'app' };
const env      = { port: null,  debug: false, workers: 0, name: null  };
const user     = { port: null,  debug: null,  workers: null, name: '' };

console.log(getConfig(user, env, defaults));
// { port: 3000, debug: false, workers: 0, name: '' }
▶ Show Solution 
function getConfig(userConfig, envConfig, defaults) {
  return Object.fromEntries(
    Object.keys(defaults).map(key => [
      key,
      (userConfig[key] ?? envConfig[key] ?? defaults[key])
    ])
  );
}

const defaults = { port: 3000, debug: false, workers: 4, name: 'app' };
const env      = { port: null,  debug: false, workers: 0, name: null  };
const user     = { port: null,  debug: null,  workers: null, name: '' };

console.log(getConfig(user, env, defaults));
// { port: 3000, debug: false, workers: 0, name: '' }
// port: user=null, env=null → default 3000 ✓
// debug: user=null, env=false → env's false preserved ✓
// workers: user=null, env=0 → env's 0 preserved ✓
// name: user='' → user's '' preserved ✓
