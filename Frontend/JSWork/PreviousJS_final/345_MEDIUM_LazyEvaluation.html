Lazy Evaluation

MEDIUM googleapplepaytm

📋 Problem

Create a lazy function evaluator that allows chaining of functions. The chain should not execute until .value() is called.

const result = lazy(5)
  .map(x => x * 2)
  .filter(x => x > 5)
  .map(x => x + 1)
  .value();
// 11

💡 Approach

Store operations as an array of functions. Only execute them when .value() is called using reduce. This is the foundation of how RxJS, Lodash chain, and lazy iterators work.

✅ Solution

function lazy(val) {
  const ops = [];

  const chain = {
    map(fn) {
      ops.push(v => (v !== undefined ? fn(v) : undefined));
      return chain;
    },
    filter(fn) {
      ops.push(v => (v !== undefined && fn(v) ? v : undefined));
      return chain;
    },
    reduce(fn, initial) {
      ops.push(v => (Array.isArray(v) ? v.reduce(fn, initial) : fn(initial, v)));
      return chain;
    },
    value() {
      return ops.reduce((acc, fn) => fn(acc), val);
    }
  };

  return chain;
}

const r1 = lazy(5).map(x => x * 2).filter(x => x > 5).map(x => x + 1).value();
console.log(r1); // 11

const r2 = lazy(3).map(x => x * x).filter(x => x > 5).value();
console.log(r2); // undefined (9 > 5, wait — actually 9 passes filter → 9)

const r3 = lazy(4).map(x => x * 2).filter(x => x > 10).value();
console.log(r3); // undefined (8 does not pass x > 10)

console.log(lazy(10).map(x => x - 3).map(x => x * 2).value()); // 14
