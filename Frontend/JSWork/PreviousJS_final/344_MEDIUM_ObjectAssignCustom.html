Custom Object.assign()

MEDIUM

📋 Problem

Recreate the functionality of Object.assign(target, ...sources). It should copy own enumerable properties from source objects into the target and return the target.

myAssign({}, {a:1}, {b:2}, {a:3}) // {a:3, b:2}
myAssign({x:0}, {x:1, y:2})       // {x:1, y:2}

💡 Approach

Iterate over each source. For each source, iterate its own enumerable keys (use hasOwnProperty or Object.keys) and copy to target.

✅ Solution

function myAssign(target, ...sources) {
  if (target == null) throw new TypeError('Cannot convert undefined or null to object');

  for (const source of sources) {
    if (source == null) continue; // skip null/undefined sources
    for (const key of Object.keys(source)) {
      target[key] = source[key];
    }
    // Also handle Symbol keys
    for (const sym of Object.getOwnPropertySymbols(source)) {
      if (Object.prototype.propertyIsEnumerable.call(source, sym)) {
        target[sym] = source[sym];
      }
    }
  }
  return target;
}

console.log(myAssign({}, {a:1}, {b:2}, {a:3})); // {a:3, b:2}
console.log(myAssign({x:0}, {x:1, y:2}));        // {x:1, y:2}
console.log(myAssign({a:1}, null, {b:2}));        // {a:1, b:2} — null skipped

// Verify it matches native
const native = Object.assign({}, {a:1,b:2}, {c:3});
const custom = myAssign({}, {a:1,b:2}, {c:3});
console.log(JSON.stringify(native) === JSON.stringify(custom)); // true

⏱ Complexity

Time: O(n·m)Space: O(1)
