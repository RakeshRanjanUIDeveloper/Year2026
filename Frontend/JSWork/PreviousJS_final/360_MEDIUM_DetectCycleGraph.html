Detect Cycle in Undirected Graph — DFS

MEDIUM amazonmicrosoftuberadobe

📋 Problem

Determine whether an undirected graph contains a cycle using DFS with parent tracking.

// Graph with cycle: 0-1, 1-2, 2-0
hasCycle(3, [[0,1],[1,2],[2,0]]) // true

// Tree (no cycle): 0-1, 1-2, 1-3
hasCycle(4, [[0,1],[1,2],[1,3]]) // false

💡 Approach

DFS from each unvisited node. Track parent to avoid going back on the same edge. If we reach an already-visited node that isn't the parent, a cycle exists.

✅ Solution

function hasCycle(numNodes, edges) {
  const adj = Array.from({ length: numNodes }, () => []);
  for (const [u, v] of edges) {
    adj[u].push(v);
    adj[v].push(u);
  }

  const visited = new Set();

  function dfs(node, parent) {
    visited.add(node);
    for (const neighbor of adj[node]) {
      if (!visited.has(neighbor)) {
        if (dfs(neighbor, node)) return true;
      } else if (neighbor !== parent) {
        return true; // visited but not parent = cycle
      }
    }
    return false;
  }

  for (let i = 0; i < numNodes; i++) {
    if (!visited.has(i) && dfs(i, -1)) return true;
  }
  return false;
}

console.log(hasCycle(3, [[0,1],[1,2],[2,0]])); // true
console.log(hasCycle(4, [[0,1],[1,2],[1,3]])); // false
console.log(hasCycle(5, [[0,1],[1,2],[2,3],[3,4],[4,1]])); // true
