Garbage Collection & Memory Leaks

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

JS uses mark-and-sweep garbage collection — unreachable objects are automatically freed. But certain patterns prevent GC from running correctly, causing memory leaks.

Common memory leak causes:

  • Global variables accidentally created (missing var/let/const)
  • Event listeners not removed when element is destroyed
  • Closures holding references to large objects longer than needed
  • Timers (setInterval) never cleared
  • Detached DOM nodes still referenced in JS variables
  • Map/Set holding object references — use WeakMap/WeakSet instead

🎯 Interview Challenges

EASY Identify the memory leak

Which of these patterns causes a memory leak? Explain why and how to fix each.

// Starter Code
// Pattern A: setInterval never cleared
function startPolling() {
  const data = new Array(10000).fill('heavy data');
  setInterval(() => {
    console.log(data.length); // data kept alive forever
  }, 1000);
}

// Pattern B: Event listener on removed element
function attachListener() {
  const btn = document.createElement('button');
  document.body.appendChild(btn);
  btn.addEventListener('click', () => console.log('clicked'));
  document.body.removeChild(btn);
  // Is this a leak?
}

// Pattern C: Closure over large object
function processLargeData() {
  const hugeArray = new Array(1000000).fill(0);
  return function getLength() {
    return hugeArray.length; // hugeArray kept in memory
  };
}
▶ Show Solution 
// Pattern A: LEAK — setInterval holds reference to 'data' via closure.
// Fix: store and clear the interval when done.
function startPolling() {
  const data = new Array(10000).fill('heavy data');
  const id = setInterval(() => { console.log(data.length); }, 1000);
  return () => clearInterval(id); // return cleanup function
}

// Pattern B: DEPENDS — Modern browsers GC detached nodes IF no JS holds a ref.
// But if you store btn in a module-level variable, it leaks.
// Fix: nullify the reference after removal.
let btn;
function attachListener() {
  btn = document.createElement('button');
  document.body.appendChild(btn);
  btn.addEventListener('click', () => console.log('clicked'));
  document.body.removeChild(btn);
  btn = null; // allow GC
}

// Pattern C: LEAK if returned function lives long and hugeArray is huge.
// Fix: only keep what you need.
function processLargeData() {
  const hugeArray = new Array(1000000).fill(0);
  const length = hugeArray.length; // capture only what's needed
  // hugeArray can now be GC'd
  return function getLength() { return length; };
}

▶ Show Solution 
const privateData = new WeakMap();

class User {
  constructor(name, password) {
    privateData.set(this, { password });
    this.name = name;
  }
  checkPassword(input) {
    return privateData.get(this).password === input;
  }
}

const u = new User('Alice', 'secret123');
console.log(u.checkPassword('secret123')); // true
console.log(u.checkPassword('wrong'));     // false
console.log(u.password);                  // undefined

// When 'u' goes out of scope:
// WeakMap → GC can collect u AND its private data (no strong ref held)
// Map     → would keep u alive forever, even if nothing else references it
HARD Build a memory-safe cache with WeakRef

Implement a cache using WeakRef so cached values can be garbage collected under memory pressure, unlike a normal Map.

// Starter Code
class WeakCache {
  constructor() {
    this.cache = new Map(); // stores WeakRef
  }

  set(key, value) {
    // store as WeakRef
  }

  get(key) {
    // deref, delete entry if GC'd, return value or undefined
  }
}

const cache = new WeakCache();
let bigObject = { data: new Array(1000).fill('x') };
cache.set('big', bigObject);
console.log(cache.get('big') !== undefined); // true
// If GC runs and bigObject is collected, cache.get('big') returns undefined
▶ Show Solution 
class WeakCache {
  constructor() { this.cache = new Map(); }

  set(key, value) {
    this.cache.set(key, new WeakRef(value));
  }

  get(key) {
    const ref = this.cache.get(key);
    if (!ref) return undefined;
    const value = ref.deref();
    if (value === undefined) {
      this.cache.delete(key); // clean up dead entry
      return undefined;
    }
    return value;
  }
}

const cache = new WeakCache();
let bigObject = { data: new Array(1000).fill('x') };
cache.set('big', bigObject);
console.log(cache.get('big') !== undefined); // true — still alive
// bigObject = null; // if this runs and GC collects it, get() returns undefined
console.log('WeakRef cache works — GC can reclaim memory under pressure');
