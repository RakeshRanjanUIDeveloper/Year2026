Sum of Consecutive Integers

MEDIUM googleamazonmicrosoftmeta

📋 Problem

Determine if a given number n can be expressed as the sum of two or more consecutive positive integers.

canBeConsecutive(15) // true  → 1+2+3+4+5 or 4+5+6 or 7+8
canBeConsecutive(9)  // true  → 2+3+4 or 4+5
canBeConsecutive(8)  // false (powers of 2 cannot)

💡 Approach

A number can be expressed as consecutive integers if and only if it is NOT a power of 2. Alternatively, try all starting points: for each start k, extend sum until it exceeds n.

✅ Solution

function canBeConsecutive(n) {
  // Math insight: n can be consecutive sum iff n is NOT a power of 2
  // Because sum of k consecutive from a = k*a + k(k-1)/2 = k(2a+k-1)/2
  // This has odd factor unless n is power of 2
  if (n <= 0) return false;
  return (n & (n - 1)) !== 0; // NOT a power of 2
}

// Explicit finder — shows which consecutive sequences work
function findConsecutive(n) {
  const results = [];
  for (let start = 1; start < n; start++) {
    let sum = 0;
    const seq = [];
    for (let k = start; sum < n; k++) {
      sum += k;
      seq.push(k);
      if (sum === n && seq.length >= 2) {
        results.push([...seq]);
        break;
      }
    }
  }
  return results;
}

console.log(canBeConsecutive(15)); // true
console.log(canBeConsecutive(9));  // true
console.log(canBeConsecutive(8));  // false
console.log(canBeConsecutive(4));  // false
console.log(findConsecutive(15));  // [[1,2,3,4,5],[4,5,6],[7,8]]
