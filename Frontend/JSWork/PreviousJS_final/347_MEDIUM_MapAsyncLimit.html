Map Async with Concurrency Limit

MEDIUM netflixubertcs

📋 Problem

Run an async function over an array with a concurrency limit. Like Promise.all but only N items run at once.

const results = await mapAsyncLimit(
  [1,2,3,4,5],
  async n => fetchUser(n),
  2           // max 2 concurrent
);
// results in original order

💡 Approach

Similar to promiseAllLimit but takes mapper fn. Maintain a pool of running promises. As each resolves, start the next. Collect results preserving index order.

✅ Solution

async function mapAsyncLimit(arr, asyncFn, limit) {
  const results = new Array(arr.length);
  let nextIndex = 0;

  async function worker() {
    while (nextIndex < arr.length) {
      const i = nextIndex++;
      results[i] = await asyncFn(arr[i], i, arr);
    }
  }

  // Spawn 'limit' concurrent workers
  await Promise.all(
    Array.from({ length: Math.min(limit, arr.length) }, worker)
  );

  return results;
}

// Demo
function fakeApiCall(id) {
  return new Promise(resolve => {
    const delay = Math.random() * 300 + 100;
    setTimeout(() => {
      console.log(`Fetched item ${id}`);
      resolve({ id, data: `result_${id}` });
    }, delay);
  });
}

mapAsyncLimit([1,2,3,4,5], fakeApiCall, 2).then(results => {
  console.log('All done:', results.map(r => r.data));
});

Time: O(n)Space: O(limit)
