📋 Problem

Simulate a mark-and-sweep garbage collector. Given a graph of objects with references, find all objects unreachable from a root set and mark them as garbage.

const heap = {
  root: { ref: ['A','B'] },
  A:    { ref: ['C'] },
  B:    { ref: [] },
  C:    { ref: ['D'] },
  D:    { ref: [] },
  E:    { ref: [] },   // unreachable!
  F:    { ref: ['E'] } // unreachable!
};
gc(heap, ['root']) // collects E, F

💡 Approach

Two phases: MARK — BFS/DFS from root set, marking all reachable objects. SWEEP — collect all objects not marked as reachable.

✅ Solution

function gc(heap, roots) {
  // MARK phase: find all reachable objects
  const reachable = new Set();
  const queue = [...roots];

  while (queue.length) {
    const id = queue.shift();
    if (reachable.has(id) || !heap[id]) continue;
    reachable.add(id);
    for (const ref of (heap[id].ref || [])) {
      if (!reachable.has(ref)) queue.push(ref);
    }
  }

  // SWEEP phase: collect unreachable objects
  const collected = [];
  const surviving = {};

  for (const id of Object.keys(heap)) {
    if (reachable.has(id)) {
      surviving[id] = heap[id];
    } else {
      collected.push(id); // garbage collected
    }
  }

  return {
    reachable: [...reachable],
    collected,
    memoryFreed: collected.length,
    surviving
  };
}

const heap = {
  root: { ref: ['A','B'] },
  A:    { ref: ['C'] },
  B:    { ref: [] },
  C:    { ref: ['D'] },
  D:    { ref: [] },
  E:    { ref: [] },   // unreachable
  F:    { ref: ['E'] } // unreachable
};

const result = gc(heap, ['root']);
console.log('Reachable:', result.reachable);
console.log('Collected (garbage):', result.collected); // ['E','F']
console.log('Memory freed:', result.memoryFreed, 'objects');

// Test with circular reference (both unreachable from root)
const heap2 = {
  root: { ref: ['A'] },
  A:    { ref: [] },
  B:    { ref: ['C'] }, // unreachable cycle
  C:    { ref: ['B'] },
};
console.log('Circular garbage:', gc(heap2, ['root']).collected); // ['B','C']
