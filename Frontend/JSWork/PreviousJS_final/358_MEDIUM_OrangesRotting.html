Oranges Rotting

MEDIUM amazonmicrosoftlinkedin

📋 Problem

Grid of oranges: 0=empty, 1=fresh, 2=rotten. Each minute rotten oranges infect adjacent fresh ones. Return minutes until all rotten, or -1 if impossible.

orangesRotting([[2,1,1],[1,1,0],[0,1,1]]) // 4
orangesRotting([[2,1,1],[0,1,1],[1,0,1]]) // -1 (isolated fresh)
orangesRotting([[0,2]])                    // 0

💡 Approach

Multi-source BFS from all initial rotten oranges simultaneously. Track fresh count. Each BFS level = 1 minute.

✅ Solution

function orangesRotting(grid) {
  const rows = grid.length, cols = grid[0].length;
  const queue = [];
  let fresh = 0;

  // Find all rotten oranges and count fresh
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      if (grid[r][c] === 2) queue.push([r, c, 0]);
      if (grid[r][c] === 1) fresh++;
    }
  }

  if (fresh === 0) return 0;

  let maxTime = 0;
  const dirs = [[-1,0],[1,0],[0,-1],[0,1]];

  while (queue.length) {
    const [r, c, time] = queue.shift();
    for (const [dr, dc] of dirs) {
      const nr = r + dr, nc = c + dc;
      if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && grid[nr][nc] === 1) {
        grid[nr][nc] = 2;
        fresh--;
        maxTime = Math.max(maxTime, time + 1);
        queue.push([nr, nc, time + 1]);
      }
    }
  }

  return fresh === 0 ? maxTime : -1;
}

console.log(orangesRotting([[2,1,1],[1,1,0],[0,1,1]])); // 4
console.log(orangesRotting([[2,1,1],[0,1,1],[1,0,1]])); // -1
console.log(orangesRotting([[0,2]]));                   // 0
