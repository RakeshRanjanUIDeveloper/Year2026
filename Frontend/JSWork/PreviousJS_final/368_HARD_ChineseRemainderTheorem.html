Chinese Remainder Theorem

HARD AtlassianTekiondropbox

📋 Problem

Solve a system of simultaneous modular congruences: x ≡ r1 (mod m1), x ≡ r2 (mod m2), ... Find the smallest non-negative x satisfying all congruences.

crt([2,3,1], [3,4,5]) // x=11
// 11 % 3 = 2 ✓, 11 % 4 = 3 ✓, 11 % 5 = 1 ✓

💡 Approach

Use the sieving (iterative) method: start with first congruence, then merge with each subsequent one. Two congruences can be merged using extended GCD to find combined solution and LCM as new modulus.

✅ Solution

function extGCD(a, b) {
  if (b === 0) return [a, 1, 0];
  const [g, x1, y1] = extGCD(b, a % b);
  return [g, y1, x1 - Math.floor(a / b) * y1];
}

function lcm(a, b) { return (a / extGCD(a, b)[0]) * b; }

function crt(remainders, moduli) {
  let x = remainders[0];
  let mod = moduli[0];

  for (let i = 1; i < remainders.length; i++) {
    const r = remainders[i];
    const m = moduli[i];
    const [g, p] = extGCD(mod, m);

    if ((r - x) % g !== 0) return null; // no solution

    const lcmVal = lcm(mod, m);
    // Merge: find new x satisfying both congruences
    x = ((x + mod * (((r - x) / g * p) % (m / g))) % lcmVal + lcmVal) % lcmVal;
    mod = lcmVal;
  }

  return x;
}

console.log(crt([2,3,1], [3,4,5]));  // 11
// Verify: 11%3=2 ✓, 11%4=3 ✓, 11%5=1 ✓

console.log(crt([0,3,4], [3,4,5]));  // 39

// No solution case (moduli not coprime and incompatible remainders)
console.log(crt([1,2], [2,4]));      // null (1%2=1 but 2%4=2, incompatible)

console.log(crt([1,1], [2,3]));      // 1 (1%2=1 ✓, 1%3=1 ✓)
