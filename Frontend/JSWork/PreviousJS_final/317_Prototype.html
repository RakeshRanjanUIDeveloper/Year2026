Prototype & Prototype Chain

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

Every JS object has an internal link [[Prototype]] (accessible via __proto__ or Object.getPrototypeOf()) pointing to another object — its prototype. When you access a property, JS searches the object first, then walks up the prototype chain until it finds it or hits null.

  • Functions have a .prototype property used when they're called as constructors
  • Object.prototype is at the top of every chain
  • Methods like .toString(), .hasOwnProperty() come from Object.prototype
function Animal(name) { this.name = name; }
Animal.prototype.speak = function() { return this.name + ' makes a sound'; };
const dog = new Animal('Rex');
dog.speak(); // found on Animal.prototype, not on dog itself

🎯 Interview Challenges

EASY hasOwnProperty vs Prototype

Given the object below, identify which properties are own vs inherited, using hasOwnProperty.

// Starter Code
function Person(name) { this.name = name; }
Person.prototype.greet = function() { return 'Hi, I am ' + this.name; };

const p = new Person('Alice');
console.log(p.hasOwnProperty('name'));   // ?
console.log(p.hasOwnProperty('greet'));  // ?
console.log('greet' in p);              // ?
▶ Show Solution 
// true  — name is set directly on p (own property)
// false — greet lives on Person.prototype, not on p itself
// true  — 'in' operator checks the full prototype chain

// Rule: hasOwnProperty = only the object itself
//       'in' operator  = object + entire chain
MEDIUM Add a Method via Prototype

Create a Rectangle constructor. Add an area() and perimeter() method on its prototype. Then create two instances and verify both share the same method reference (not copies).

// Starter Code
function Rectangle(w, h) {
  // your code
}

// add area() and perimeter() on prototype

const r1 = new Rectangle(4, 5);
const r2 = new Rectangle(3, 7);
console.log(r1.area());       // 20
console.log(r2.perimeter());  // 20
console.log(r1.area === r2.area); // true — shared, not copied
▶ Show Solution 
function Rectangle(w, h) {
  this.width = w;
  this.height = h;
}

Rectangle.prototype.area = function() { return this.width * this.height; };
Rectangle.prototype.perimeter = function() { return 2 * (this.width + this.height); };

const r1 = new Rectangle(4, 5);
const r2 = new Rectangle(3, 7);
console.log(r1.area());             // 20
console.log(r2.perimeter());        // 20
console.log(r1.area === r2.area);   // true — one function on the prototype, shared
HARD Implement Prototypal Inheritance from Scratch

Implement Animal → Dog inheritance using only prototypes (no class keyword). Dog should inherit speak() from Animal but override it. Verify instanceof works correctly.

// Starter Code
function Animal(name) { this.name = name; }
Animal.prototype.speak = function() { return this.name + ' makes a sound'; };

function Dog(name, breed) {
  // call Animal constructor
  this.breed = breed;
}

// set up prototype chain so Dog inherits Animal
// override speak() on Dog.prototype

const d = new Dog('Rex', 'Labrador');
console.log(d.speak());        // 'Rex barks'
console.log(d instanceof Dog);    // true
console.log(d instanceof Animal); // true
▶ Show Solution 
function Animal(name) { this.name = name; }
Animal.prototype.speak = function() { return this.name + ' makes a sound'; };

function Dog(name, breed) {
  Animal.call(this, name); // inherit own properties
  this.breed = breed;
}

Dog.prototype = Object.create(Animal.prototype); // set up chain
Dog.prototype.constructor = Dog;                 // fix constructor ref
Dog.prototype.speak = function() { return this.name + ' barks'; }; // override

const d = new Dog('Rex', 'Labrador');
console.log(d.speak());           // 'Rex barks'
console.log(d instanceof Dog);    // true
console.log(d instanceof Animal); // true
