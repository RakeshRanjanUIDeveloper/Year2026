Implement Promise.race()

MEDIUM netflixreddittcs

📋 Problem

Polyfill Promise.race() — resolves or rejects as soon as the first promise in the iterable settles.

myPromiseRace([
  new Promise(r => setTimeout(() => r('slow'), 500)),
  new Promise(r => setTimeout(() => r('fast'), 100)),
]).then(v => console.log(v)); // 'fast'

💡 Approach

Return a new Promise. Iterate all input promises and call .then/.catch on each. The first one to settle wins and resolves/rejects the outer promise.

✅ Solution

function myPromiseRace(promises) {
  return new Promise((resolve, reject) => {
    for (const p of promises) {
      Promise.resolve(p).then(resolve, reject);
      // First settlement wins; subsequent ones are ignored
      // because a Promise can only be settled once
    }
  });
}

// Demo
const p1 = new Promise(r => setTimeout(() => r('slow'), 500));
const p2 = new Promise(r => setTimeout(() => r('fast'), 100));
const p3 = new Promise((_, r) => setTimeout(() => r('error'), 200));

myPromiseRace([p1, p2]).then(v => console.log('Winner:', v)); // 'fast'

// Race with rejection
const p4 = new Promise(r => setTimeout(() => r('ok'), 300));
const p5 = new Promise((_, r) => setTimeout(() => r(new Error('fail')), 50));
myPromiseRace([p4, p5]).catch(e => console.log('First reject:', e.message)); // 'fail'

// Also implement Promise.any (resolves on FIRST success, ignores rejections)
function myPromiseAny(promises) {
  return new Promise((resolve, reject) => {
    let rejCount = 0;
    const errors = [];
    promises.forEach((p, i) => {
      Promise.resolve(p)
        .then(resolve)
        .catch(e => {
          errors[i] = e;
          if (++rejCount === promises.length)
            reject(new AggregateError(errors, 'All promises rejected'));
        });
    });
  });
}
