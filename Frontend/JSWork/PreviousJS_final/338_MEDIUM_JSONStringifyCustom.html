Custom JSON.stringify

MEDIUM adobestripepaytm

📋 Problem

Implement a custom version of JSON.stringify that handles: strings, numbers, booleans, null, arrays, and objects. Skip undefined, functions, and symbols.

myStringify({ a: 1, b: [2, true, null, "hi"] })
// '{"a":1,"b":[2,true,null,"hi"]}'

myStringify(undefined) // undefined
myStringify(() => {})  // undefined

💡 Approach

Recursively handle each type. Use typeof checks. Arrays: wrap in [], join with commas. Objects: wrap in {}, map entries to 'key':value pairs, skip undefined/function/symbol values.

✅ Solution

function myStringify(value) {
  // Primitives
  if (value === null)            return 'null';
  if (value === undefined)       return undefined;
  if (typeof value === 'function' || typeof value === 'symbol') return undefined;
  if (typeof value === 'boolean') return String(value);
  if (typeof value === 'number') {
    if (!isFinite(value)) return 'null'; // NaN, Infinity → null
    return String(value);
  }
  if (typeof value === 'string') return `"${value.replace(/\/g, '\\').replace(/"/g, '\"')}"`;

  // Array
  if (Array.isArray(value)) {
    const items = value.map(item => {
      const s = myStringify(item);
      return s === undefined ? 'null' : s; // undefined in arrays → null
    });
    return '[' + items.join(',') + ']';
  }

  // Object
  if (typeof value === 'object') {
    const pairs = [];
    for (const key of Object.keys(value)) {
      const v = myStringify(value[key]);
      if (v !== undefined) pairs.push(`"${key}":${v}`);
    }
    return '{' + pairs.join(',') + '}';
  }
}

console.log(myStringify({ a: 1, b: [2, true, null, "hi"] }));
console.log(myStringify([1, undefined, 3]));  // [1,null,3]
console.log(myStringify({ fn: () => {}, x: 1 })); // {"x":1}
console.log(myStringify(null));               // null
console.log(myStringify(undefined));          // undefined
