Promise.all With Concurrency Limit

MEDIUM netflixuberpaytm

📋 Problem

Implement a function that works like Promise.all() but runs at most N promises concurrently at any time.

const results = await promiseAllLimit(tasks, 2);
// runs max 2 tasks at a time, returns all results in order

💡 Approach

Use a pool pattern. Start up to 'limit' tasks. When one finishes, start the next. Track results by index to maintain order.

✅ Solution

async function promiseAllLimit(tasks, limit) {
  const results = new Array(tasks.length);
  let index = 0;

  async function runNext() {
    const i = index++;
    if (i >= tasks.length) return;
    results[i] = await tasks[i]();
    await runNext(); // pick up next task
  }

  // Start 'limit' workers
  const workers = Array.from({ length: Math.min(limit, tasks.length) }, runNext);
  await Promise.all(workers);
  return results;
}

// Demo
function makeTask(id, ms) {
  return () => new Promise(resolve => {
    console.log(`Task ${id} started`);
    setTimeout(() => { console.log(`Task ${id} done`); resolve(id); }, ms);
  });
}

const tasks = [
  makeTask(1, 500), makeTask(2, 300),
  makeTask(3, 400), makeTask(4, 200)
];

promiseAllLimit(tasks, 2).then(results => {
  console.log('All results:', results); // [1, 2, 3, 4] in order
});

Time: O(n)Space: O(limit)
