Event Propagation & Event Listener

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

When an event fires on a DOM element, it travels in 3 phases:

  1. Capture phase — from window down to the target
  2. Target phase — the element itself
  3. Bubble phase — from target back up to window
  • addEventListener(event, fn, true) — capture phase
  • addEventListener(event, fn, false) — bubble phase (default)
  • event.stopPropagation() — stop travelling
  • event.preventDefault() — prevent default browser action
  • event.target — element that triggered the event
  • event.currentTarget — element the listener is attached to

document.getElementById('btn').addEventListener('click', (e) => {
  e.stopPropagation(); // stops bubble → outer click never fires

▶ Show Solution 
// 1. { once: true } — browser auto-removes after first call
btn.addEventListener('click', () => console.log('fired once!'), { once: true });

// 2. removeEventListener needs the SAME function reference
//    arrow functions create new refs each time — this is why named functions matter
function handleClick() { console.log('removable'); }
btn.addEventListener('click', handleClick);
btn.removeEventListener('click', handleClick); // ✅ works — same reference

// 3. passive: true tells browser "this listener won't call preventDefault()"
//    Browser can then scroll immediately without waiting for JS → smoother scroll
//    Never call e.preventDefault() inside a passive listener — it's ignored
HARD Build Event Emitter (pub/sub) from scratch

Implement a custom EventEmitter class with on(event, fn), off(event, fn), emit(event, ...args), and once(event, fn).

// Starter Code

  // implement on, off, emit, once

const emitter = new EventEmitter();
const handler = (msg) => console.log('received:', msg);

emitter.on('message', handler);
emitter.emit('message', 'hello');   // received: hello
emitter.emit('message', 'world');   // received: world
emitter.off('message', handler);
emitter.emit('message', 'gone');    // nothing

emitter.once('ping', () => console.log('pong'));
emitter.emit('ping'); // pong
emitter.emit('ping'); // nothing
▶ Show Solution 
class EventEmitter {
  constructor() { this._events = {}; }

  on(event, fn) {
    (this._events[event] = this._events[event] || []).push(fn);
    return this;
  }

  off(event, fn) {
    if (this._events[event])
      this._events[event] = this._events[event].filter(f => f !== fn && f._original !== fn);
    return this;
  }

  emit(event, ...args) {
    (this._events[event] || []).slice().forEach(fn => fn(...args));
    return this;
  }

  once(event, fn) {
    const wrapper = (...args) => { fn(...args); this.off(event, wrapper); };
    wrapper._original = fn;
    return this.on(event, wrapper);
  }
}

const emitter = new EventEmitter();
const handler = (msg) => console.log('received:', msg);
emitter.on('message', handler);
emitter.emit('message', 'hello');
emitter.emit('message', 'world');
emitter.off('message', handler);
emitter.emit('message', 'gone');
emitter.once('ping', () => console.log('pong'));
emitter.emit('ping');
emitter.emit('ping');
