JSON.parse() and JSON.stringify()

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

JSON (JavaScript Object Notation) is a text format for data interchange.

  • JSON.stringify(value, replacer, space) — converts JS → JSON string
  • JSON.parse(text, reviver) — converts JSON string → JS value
  • Limitations: undefined, functions, Symbols are dropped; circular refs throw; Date becomes string
  • replacer — filter/transform on stringify
  • reviver — transform on parse (great for reviving Dates)
JSON.stringify({ a: 1, fn: () => {} }); // '{"a":1}' — fn dropped!
JSON.parse('{"date":"2024-01-01"}');      // date is a string, not Date object

▶ Show Solution 
// undefined         → undefined (not a valid JSON value)
// {"b":1}           → undefined keys are omitted from objects
// [null,1,null]     → undefined in arrays becomes null
// {}                → functions are dropped
// null              → NaN and Infinity become null in JSON
// null

// Key rule: JSON only supports: string, number, boolean, null, array, object
MEDIUM Replacer and Reviver

Use a replacer to exclude passwords from JSON output, and a reviver to restore Date objects from strings.

// Starter Code
const user = {
  name: 'Alice',
  password: 'secret',
  createdAt: new Date('2024-01-15'),
  age: 25
};

// 1. Stringify but exclude 'password'
const safe = JSON.stringify(user, /* replacer */, 2);
console.log(safe); // no password field

// 2. Parse back and restore Date
const parsed = JSON.parse(safe, /* reviver */);
console.log(parsed.createdAt instanceof Date); // true
▶ Show Solution 
const user = {
  name: 'Alice', password: 'secret',
  createdAt: new Date('2024-01-15'), age: 25
};

// Replacer: return undefined to exclude a key
const safe = JSON.stringify(user, (key, value) => {
  if (key === 'password') return undefined;
  return value;
}, 2);
console.log(safe); // no password

// Reviver: detect ISO date strings and convert to Date
const parsed = JSON.parse(safe, (key, value) => {
  if (typeof value === 'string' && /^\d{4}-\d{2}-\d{2}T/.test(value))
    return new Date(value);
  return value;
});
console.log(parsed.createdAt instanceof Date); // true
console.log(parsed.createdAt.getFullYear());   // 2024

▶ Show Solution 
function safeStringify(obj) {
  const seen = new WeakSet();
  return JSON.stringify(obj, (key, value) => {
    if (typeof value === 'object' && value !== null) {
      if (seen.has(value)) return '[Circular]';
      seen.add(value);
    }
    return value;
  });
}

function jsonDeepClone(obj) {
  // Fast for plain objects (drops undefined, functions, symbols)
  return JSON.parse(JSON.stringify(obj));
}

const a = { x: 1 };
a.self = a;
console.log(safeStringify(a)); // '{"x":1,"self":"[Circular]"}'

const original = { a: 1, b: { c: 2 }, arr: [1, 2] };
const clone = jsonDeepClone(original);
clone.b.c = 99;
console.log(original.b.c); // 2
