Pipe and Compose

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

Pipe and Compose are functional programming patterns for chaining multiple functions together.

  • compose(f, g, h)(x) — applies right to left: f(g(h(x)))
  • pipe(f, g, h)(x) — applies left to right: h(g(f(x)))
  • pipe is more intuitive (reads in execution order)
  • Both take any number of functions and return a new function
const double = x => x * 2;
const addOne = x => x + 1;
const square = x => x * x;
pipe(double, addOne, square)(3); // square(addOne(double(3))) = square(7) = 49

🎯 Interview Challenges

EASY Implement pipe and compose

Implement both pipe(...fns) and compose(...fns). Both return a function that passes a value through the chain.

// Starter Code
function pipe(...fns) {
  // left to right
}

function compose(...fns) {
  // right to left
}

const double = x => x * 2;
const addOne = x => x + 1;
const square = x => x * x;

console.log(pipe(double, addOne, square)(3));    // (3*2+1)^2 = 49
console.log(compose(square, addOne, double)(3)); // same result: 49
▶ Show Solution 
function pipe(...fns) {
  return (value) => fns.reduce((acc, fn) => fn(acc), value);
}

function compose(...fns) {
  return (value) => fns.reduceRight((acc, fn) => fn(acc), value);
}

const double = x => x * 2;
const addOne = x => x + 1;
const square = x => x * x;

console.log(pipe(double, addOne, square)(3));    // 49
console.log(compose(square, addOne, double)(3)); // 49
MEDIUM Async pipe

Extend pipe to handle async functions — each step can be sync or async, and the chain awaits each one.

// Starter Code
function pipeAsync(...fns) {
  // each fn may return a Promise
}

const fetchUser    = async id  => ({ id, name: 'Alice' });
const fetchOrders  = async user => ({ ...user, orders: [1, 2, 3] });
const formatOutput = user => `${user.name} has ${user.orders.length} orders`;

const processUser = pipeAsync(fetchUser, fetchOrders, formatOutput);
processUser(1).then(console.log); // 'Alice has 3 orders'
▶ Show Solution 
function pipeAsync(...fns) {
  return (value) => fns.reduce(async (acc, fn) => fn(await acc), Promise.resolve(value));
}

const fetchUser    = async id   => ({ id, name: 'Alice' });
const fetchOrders  = async user => ({ ...user, orders: [1, 2, 3] });
const formatOutput = user => `${user.name} has ${user.orders.length} orders`;

const processUser = pipeAsync(fetchUser, fetchOrders, formatOutput);
processUser(1).then(console.log); // 'Alice has 3 orders'
HARD Data transformation pipeline

Build a real-world data pipeline using pipe. Transform raw API data: parse → validate → normalize → format — each as a separate composable function.

// Starter Code
const rawData = [
  { name: '  alice ', age: '25', email: 'ALICE@TEST.COM' },
  { name: 'BOB',     age: '17', email: 'bob@test.com' },
  { name: ' carol ', age: '30', email: 'carol@test.com' },
];

const trim        = users => users.map(u => ({ ...u, name: u.name.trim() }));
const lowercase   = users => users.map(u => ({ ...u, email: u.email.toLowerCase() }));
const parseAge    = users => users.map(u => ({ ...u, age: parseInt(u.age) }));
const filterAdult = users => users.filter(u => u.age >= 18);
const formatNames = users => users.map(u => ({ ...u, name: u.name[0].toUpperCase() + u.name.slice(1).toLowerCase() }));

function pipe(...fns) { return v => fns.reduce((a,f) => f(a), v); }

const processUsers = pipe(trim, lowercase, parseAge, filterAdult, formatNames);
console.log(processUsers(rawData));
// [ { name: 'Alice', age: 25, email: 'alice@test.com' },
//   { name: 'Carol', age: 30, email: 'carol@test.com' } ]
