Cookies

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

Cookies are small key-value strings stored in the browser and sent with every HTTP request to the same domain. Unlike localStorage they have expiry, domain scope, and security flags.

  • Set: document.cookie = "key=value; expires=...; path=/"
  • Read: document.cookie returns ALL cookies as one string
  • Delete: set the same cookie with an expired date
  • Flags: Secure (HTTPS only), HttpOnly (JS can't read — server-set), SameSite

function setCookie(name, value, days) {
  const expires = new Date(Date.now() + days * 864e5).toUTCString();
  document.cookie = `${encodeURIComponent(name)}=${encodeURIComponent(value)}; expires=${expires}; path=/`;
}

function getCookie(name) {
  const key = encodeURIComponent(name) + '=';
  const found = document.cookie.split(';').map(c => c.trim()).find(c => c.startsWith(key));
  return found ? decodeURIComponent(found.slice(key.length)) : null;
}

function deleteCookie(name) {
  document.cookie = `${encodeURIComponent(name)}=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/`;
}

setCookie('user', 'Alice', 7);
console.log(getCookie('user')); // 'Alice'
deleteCookie('user');
console.log(getCookie('user')); // null

function setCookie(name, value, days=30) {
  const e = new Date(Date.now() + days*864e5).toUTCString();
  document.cookie = `${name}=${value}; expires=${e}; path=/`;
}
function getCookie(name) {
  const c = document.cookie.split(';').map(s=>s.trim()).find(s=>s.startsWith(name+'='));
  return c ? c.split('=')[1] : null;
}

function savePreferences(theme, locale) {
  localStorage.setItem('theme', theme); // stays on client
  setCookie('locale', locale);          // sent to server on every request
}

function loadPreferences() {
  return {
    theme: localStorage.getItem('theme'),
    locale: getCookie('locale')
  };
}

savePreferences('dark', 'en-US');
console.log(loadPreferences()); // { theme: 'dark', locale: 'en-US' }

const CookieManager = {
  set(name, value, days = 7) {
    const expires = new Date(Date.now() + days * 864e5).toUTCString();
    const encoded = encodeURIComponent(JSON.stringify(value));
    document.cookie = `${name}=${encoded}; expires=${expires}; path=/; SameSite=Strict`;
  },
  get(name) {
    const c = document.cookie.split(';').map(s=>s.trim()).find(s=>s.startsWith(name+'='));
    if (!c) return null;
    try { return JSON.parse(decodeURIComponent(c.slice(name.length + 1))); }
    catch { return null; }
  },
  remove(name) {
    document.cookie = `${name}=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/`;
  },
  getAll() {
    return Object.fromEntries(
      document.cookie.split(';')
        .map(c => c.trim().split('='))
        .filter(([k]) => k)
        .map(([k, v]) => {
          try { return [k, JSON.parse(decodeURIComponent(v))]; }
          catch { return [k, decodeURIComponent(v)]; }
        })
    );
  }
};

CookieManager.set('prefs', { theme: 'dark', lang: 'en' });
console.log(CookieManager.get('prefs')); // { theme: 'dark', lang: 'en' }
console.log(CookieManager.getAll());
