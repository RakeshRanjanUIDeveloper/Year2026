Modules — import / export / Dynamic Import

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

ES Modules let you split code into reusable files. Browsers support them natively with type="module".

  • Named export: export const fn = () => {} — import as import { fn } from './file.js'
  • Default export: export default class MyClass {} — import as import MyClass from './file.js'
  • Re-export: export { fn } from './other.js'
  • Dynamic import: const mod = await import('./heavy.js') — loads on demand
  • Modules are always strict mode, run once and are cached
// math.js
export const add = (a,b) => a + b;
export default function multiply(a,b) { return a*b; }

// app.js
import multiply, { add } from './math.js';
import('./heavy.js').then(mod => mod.init()); // dynamic

MEDIUM Dynamic Import for code splitting

Show how dynamic import() is used for lazy loading — only load a heavy module when actually needed (e.g. on button click).

// Starter Code
// Simulate a heavy module
const heavyModule = {
  processData: (data) => data.map(x => x * 2)
};

// Without dynamic import — loaded even if never used:
// import { processData } from './heavy.js';

// With dynamic import — only loaded on demand:
async function onButtonClick(data) {
  // dynamically import only when needed
  const { processData } = await import('./heavy.js'); // in real code
  // For demo, simulate the dynamic load:
  const mod = await Promise.resolve(heavyModule);
  return mod.processData(data);
}

onButtonClick([1, 2, 3]).then(result => console.log(result)); // [2, 4, 6]

  }

// 3. Load by route (React-like code splitting)

const plugins = {
  logger:    { name: 'logger',    init(ctx) { ctx.log = console.log; console.log('Logger ready'); }},
  analytics: { name: 'analytics', init(ctx) { ctx.track = e => console.log('track:', e); console.log('Analytics ready'); }},
};

class PluginManager {
  constructor() { this.context = {}; this.loaded = new Set(); }

  async load(name) {
    if (this.loaded.has(name)) return; // cached — don't load twice
    const plugin = await Promise.resolve(plugins[name]); // simulate dynamic import
    if (!plugin) throw new Error(`Plugin "${name}" not found`);
    plugin.init(this.context);
    this.loaded.add(name);
  }

  async loadAll(names) {
    await Promise.all(names.map(n => this.load(n)));
  }
}

const pm = new PluginManager();
pm.loadAll(['logger', 'analytics']).then(() => {
  pm.context.log('Hello');
  pm.context.track('page_view');
});
