AbortController

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

AbortController lets you cancel in-flight fetch requests (or any async operation that accepts a signal).

  • Create: const controller = new AbortController();
  • Pass signal: fetch(url, { signal: controller.signal })
  • Cancel: controller.abort()
  • On abort, fetch throws a DOMException with name === 'AbortError'
const controller = new AbortController();
setTimeout(() => controller.abort(), 2000); // cancel after 2s
fetch(url, { signal: controller.signal })
  .then(res => res.json())
  .catch(err => {
    if (err.name === 'AbortError') console.log('Request cancelled');
  });

▶ Show Solution 
async function fetchWithTimeout(url, ms) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), ms);
  try {
    const res = await fetch(url, { signal: controller.signal });
    clearTimeout(timer);
    return await res.json();
  } catch (err) {
    if (err.name === 'AbortError') throw new Error(`Request timed out after ${ms}ms`);
    throw err;
  }
}

fetchWithTimeout('https://jsonplaceholder.typicode.com/posts/1', 5000)
  .then(data => console.log(data.title))
  .catch(err => console.log('Error:', err.message));
MEDIUM Cancel previous request on new input (search pattern)

Build a search function that cancels the previous API call whenever a new search term is typed. This is the real-world debounce + AbortController pattern.

// Starter Code
function createSearch() {
  // return a search(term) function that cancels previous call
}

const search = createSearch();
search('java');       // cancelled immediately by next call
search('javascript'); // this one should complete
▶ Show Solution 
function createSearch() {
  let controller = null;
  return async function search(term) {
    if (controller) controller.abort(); // cancel previous
    controller = new AbortController();
    try {
      const res = await fetch(
        `https://jsonplaceholder.typicode.com/posts?q=${term}`,
        { signal: controller.signal }
      );
      const data = await res.json();
      console.log(`Results for "${term}":`, data.length);
    } catch (err) {
      if (err.name === 'AbortError') console.log(`Cancelled: "${term}"`);
      else console.error(err);
    }
  };
}

const search = createSearch();
search('java');
search('javascript');

const { promise, abort } = makeAbortable(longTask);

  .then(v => console.log('Result:', v))
  .catch(e => console.log('Aborted:', e.message));

▶ Show Solution 
function makeAbortable(promise) {
  let rejectFn;
  const wrapper = new Promise((resolve, reject) => {
    rejectFn = reject;
    promise.then(resolve, reject);
  });
  return {
    promise: wrapper,
    abort() {
      const err = new DOMException('Aborted', 'AbortError');
      rejectFn(err);
    }
  };
}

const longTask = new Promise(resolve => setTimeout(() => resolve('done'), 3000));
const { promise, abort } = makeAbortable(longTask);

promise
  .then(v => console.log('Result:', v))
  .catch(e => console.log('Aborted:', e.message)); // Aborted: Aborted

setTimeout(abort, 1000);
