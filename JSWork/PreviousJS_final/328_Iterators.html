Iterators & Iteration Protocol

JavaScript Interview Topic — Easy · Medium · Hard Challenges

📖 Concept

An iterator is any object with a next() method returning { value, done }. An iterable is any object with [Symbol.iterator]() that returns an iterator.

  • Built-in iterables: Array, String, Map, Set, arguments, NodeList
  • for...of, spread ..., destructuring all use the iterator protocol internally
  • You can make any object iterable by implementing [Symbol.iterator]
const iter = [1,2,3][Symbol.iterator]();
iter.next(); // { value: 1, done: false }
iter.next(); // { value: 2, done: false }
iter.next(); // { value: 3, done: false }
iter.next(); // { value: undefined, done: true }

Create a counter iterator that counts from start to end, step by step, using the iterator protocol manually.

function makeCounter(start, end) {

  return {

      // return { value, done }
    }
  };
}

const counter = makeCounter(1, 3);
console.log(counter.next()); // { value: 1, done: false }
console.log(counter.next()); // { value: 2, done: false }

console.log(counter.next()); // { value: undefined, done: true }
▶ Show Solution 
function makeCounter(start, end) {
  let current = start;
  return {
    next() {
      return current <= end
        ? { value: current++, done: false }
        : { value: undefined, done: true };
    }
  };
}

const counter = makeCounter(1, 3);
console.log(counter.next()); // { value: 1, done: false }
console.log(counter.next()); // { value: 2, done: false }
console.log(counter.next()); // { value: 3, done: false }

MEDIUM Make a custom class iterable

Make a LinkedList class iterable so it can be used with for...of and spread operator.

// Starter Code
class Node {
  constructor(val, next = null) { this.val = val; this.next = next; }
}

class LinkedList {
  constructor() { this.head = null; }
  push(val) {
    this.head = new Node(val, this.head);
    return this;
  }
  [Symbol.iterator]() {
    // traverse from head
  }
}

const list = new LinkedList();
list.push(3).push(2).push(1);
for (const val of list) console.log(val); // 1, 2, 3
console.log([...list]); // [1, 2, 3]
▶ Show Solution 
class Node { constructor(val, next=null) { this.val=val; this.next=next; } }

class LinkedList {
  constructor() { this.head = null; }
  push(val) { this.head = new Node(val, this.head); return this; }
  [Symbol.iterator]() {
    let current = this.head;
    return {
      next() {
        if (current) { const v = current.val; current = current.next; return { value: v, done: false }; }
        return { value: undefined, done: true };
      }
    };
  }
}

const list = new LinkedList();
list.push(3).push(2).push(1);
for (const val of list) console.log(val);
console.log([...list]);

function fibIterator() {
  let a = 0, b = 1;
  return {
    next() { const val = a; [a, b] = [b, a + b]; return { value: val, done: false }; },
    return(val) { console.log('iterator closed'); return { value: val, done: true }; },
    [Symbol.iterator]() { return this; }
  };
}

function takeN(iterator, n) {
  const result = [];
  for (const val of iterator) {
    result.push(val);
    if (result.length === n) break; // triggers iterator.return()
  }
  return result;
}

const fibs = fibIterator();
console.log(takeN(fibs, 8)); // [0,1,1,2,3,5,8,13]
