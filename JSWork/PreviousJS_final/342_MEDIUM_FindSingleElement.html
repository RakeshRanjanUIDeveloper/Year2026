Find Single Element in Sorted Array

MEDIUM amazonapplenetflix

📋 Problem

Given a sorted array where every element appears exactly twice except one, find that single element. Must run in O(log n) time.

singleNonDuplicate([1,1,2,3,3,4,4,8,8]) // 2
singleNonDuplicate([3,3,7,7,10,11,11])   // 10

💡 Approach

Binary search on even indices. Before the single element, pairs start at even indices. After it, pairs start at odd indices. Use this to shrink the search space.

✅ Solution

function singleNonDuplicate(nums) {
  let left = 0, right = nums.length - 1;

  while (left < right) {
    let mid = Math.floor((left + right) / 2);
    // Ensure mid is even for consistent pair comparison
    if (mid % 2 === 1) mid--;

    if (nums[mid] === nums[mid + 1]) {
      // Pair is intact — single element is to the right
      left = mid + 2;
    } else {
      // Pair is broken — single element is here or to the left
      right = mid;
    }
  }
  return nums[left];
}

console.log(singleNonDuplicate([1,1,2,3,3,4,4,8,8])); // 2
console.log(singleNonDuplicate([3,3,7,7,10,11,11]));   // 10
console.log(singleNonDuplicate([1]));                   // 1
